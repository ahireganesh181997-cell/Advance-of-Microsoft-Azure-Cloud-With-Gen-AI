#!/bin/bash

set -euo pipefail

# ============================================================
# Azure CLI - IT Production Environment
# Region: Central India
# ============================================================

# -----------------------------
# Variables
# -----------------------------
RG="RG-IT-PROD"
LOCATION="centralindia"

VNET="VNET-IT-PROD"
VNET_ADDRESS="192.168.0.0/16"

WEB_SUBNET="WEB-SUBNET-IT-PROD"
WEB_SUBNET_PREFIX="192.168.0.0/24"
WEB_NSG="WEB-SUBNET-NSG"
WEB_PIP="WEB-VM-IT-PROD-PIP"
WEB_NIC="WEB-VM-IT-PROD-NIC"
WEB_VM="WEB-VM-IT-PROD"
WEB_OS_DISK="WEB-VM-IT-PROD-OS-DISK"

DB_SUBNET="DB-SUBNET-IT-PROD"
DB_SUBNET_PREFIX="192.168.1.0/24"
DB_NSG="DB-SUBNET-NSG"
DB_PIP="DB-VM-IT-PROD-PIP"
DB_NIC="DB-VM-IT-PROD-NIC"
DB_VM="DB-VM-IT-PROD"
DB_OS_DISK="DB-VM-IT-PROD-OS-DISK"

VM_SIZE="Standard_B2s_v2"

IMAGE="MicrosoftWindowsServer:WindowsServer:2025-datacenter-azure-edition:latest"

ADMIN_USERNAME="azureuser"
ADMIN_PASSWORD='Azure#123456789'

# ============================================================
# 1. Resource Group
# ============================================================

echo "Creating Resource Group..."

az group create \
    --name "$RG" \
    --location "$LOCATION" \
    --output none

# ============================================================
# 2. Virtual Network
# ============================================================

echo "Creating Virtual Network..."

az network vnet create \
    --resource-group "$RG" \
    --name "$VNET" \
    --location "$LOCATION" \
    --address-prefixes "$VNET_ADDRESS" \
    --output none

# ============================================================
# 3. Create WEB NSG
# ============================================================

echo "Creating WEB NSG..."

az network nsg create \
    --resource-group "$RG" \
    --name "$WEB_NSG" \
    --location "$LOCATION" \
    --output none

# ============================================================
# 4. Create DB NSG
# ============================================================

echo "Creating DB NSG..."

az network nsg create \
    --resource-group "$RG" \
    --name "$DB_NSG" \
    --location "$LOCATION" \
    --output none

# ============================================================
# 5. Create WEB Subnet
#    NSG attached ONLY to subnet
# ============================================================

echo "Creating WEB Subnet..."

az network vnet subnet create \
    --resource-group "$RG" \
    --vnet-name "$VNET" \
    --name "$WEB_SUBNET" \
    --address-prefixes "$WEB_SUBNET_PREFIX" \
    --network-security-group "$WEB_NSG" \
    --output none

# ============================================================
# 6. Create DB Subnet
#    NSG attached ONLY to subnet
# ============================================================

echo "Creating DB Subnet..."

az network vnet subnet create \
    --resource-group "$RG" \
    --vnet-name "$VNET" \
    --name "$DB_SUBNET" \
    --address-prefixes "$DB_SUBNET_PREFIX" \
    --network-security-group "$DB_NSG" \
    --output none

# ============================================================
# 7. WEB NSG - Allow RDP 3389
# ============================================================

echo "Creating WEB RDP NSG Rule..."

az network nsg rule create \
    --resource-group "$RG" \
    --nsg-name "$WEB_NSG" \
    --name "Allow-RDP-3389" \
    --priority 100 \
    --direction Inbound \
    --access Allow \
    --protocol Tcp \
    --source-address-prefixes "*" \
    --source-port-ranges "*" \
    --destination-address-prefixes "*" \
    --destination-port-ranges 3389 \
    --output none

# ============================================================
# 8. DB NSG - Allow RDP 3389
# ============================================================

echo "Creating DB RDP NSG Rule..."

az network nsg rule create \
    --resource-group "$RG" \
    --nsg-name "$DB_NSG" \
    --name "Allow-RDP-3389" \
    --priority 100 \
    --direction Inbound \
    --access Allow \
    --protocol Tcp \
    --source-address-prefixes "*" \
    --source-port-ranges "*" \
    --destination-address-prefixes "*" \
    --destination-port-ranges 3389 \
    --output none

# ============================================================
# 9. WEB Public IP
# ============================================================

echo "Creating WEB Public IP..."

az network public-ip create \
    --resource-group "$RG" \
    --name "$WEB_PIP" \
    --location "$LOCATION" \
    --sku Standard \
    --allocation-method Static \
    --output none

# ============================================================
# 10. DB Public IP
# ============================================================

echo "Creating DB Public IP..."

az network public-ip create \
    --resource-group "$RG" \
    --name "$DB_PIP" \
    --location "$LOCATION" \
    --sku Standard \
    --allocation-method Static \
    --output none

# ============================================================
# 11. WEB NIC
#     IMPORTANT: No NSG attached directly to NIC
# ============================================================

echo "Creating WEB NIC..."

az network nic create \
    --resource-group "$RG" \
    --name "$WEB_NIC" \
    --location "$LOCATION" \
    --vnet-name "$VNET" \
    --subnet "$WEB_SUBNET" \
    --public-ip-address "$WEB_PIP" \
    --output none

# ============================================================
# 12. DB NIC
#     IMPORTANT: No NSG attached directly to NIC
# ============================================================

echo "Creating DB NIC..."

az network nic create \
    --resource-group "$RG" \
    --name "$DB_NIC" \
    --location "$LOCATION" \
    --vnet-name "$VNET" \
    --subnet "$DB_SUBNET" \
    --public-ip-address "$DB_PIP" \
    --output none

# ============================================================
# 13. Verify NICs have NO direct NSG
# ============================================================

echo "Verifying NIC NSG configuration..."

WEB_NIC_NSG=$(az network nic show \
    --resource-group "$RG" \
    --name "$WEB_NIC" \
    --query "networkSecurityGroup.id" \
    --output tsv)

DB_NIC_NSG=$(az network nic show \
    --resource-group "$RG" \
    --name "$DB_NIC" \
    --query "networkSecurityGroup.id" \
    --output tsv)

if [ -n "$WEB_NIC_NSG" ]; then
    echo "ERROR: $WEB_NIC has a directly attached NSG."
    exit 1
fi

if [ -n "$DB_NIC_NSG" ]; then
    echo "ERROR: $DB_NIC has a directly attached NSG."
    exit 1
fi

# ============================================================
# 14. VM Creation Function
#     Zone 1 -> Zone 2 -> Zone 3
# ============================================================

create_vm_with_retry() {

    local VM_NAME="$1"
    local NIC_NAME="$2"
    local OS_DISK_NAME="$3"

    echo ""
    echo "============================================================"
    echo "Creating VM: $VM_NAME"
    echo "============================================================"

    for ZONE in 1 2 3
    do

        echo ""
        echo "Attempting $VM_NAME in Availability Zone $ZONE..."

        # ----------------------------------------------------
        # Remove failed VM if present
        # ----------------------------------------------------

        if az vm show \
            --resource-group "$RG" \
            --name "$VM_NAME" \
            --output none 2>/dev/null
        then

            echo "Removing previously created failed VM..."

            az vm delete \
                --resource-group "$RG" \
                --name "$VM_NAME" \
                --yes \
                --output none
        fi

        # ----------------------------------------------------
        # Remove failed OS disk if present
        # ----------------------------------------------------

        if az disk show \
            --resource-group "$RG" \
            --name "$OS_DISK_NAME" \
            --output none 2>/dev/null
        then

            echo "Removing previously created failed OS disk..."

            az disk delete \
                --resource-group "$RG" \
                --name "$OS_DISK_NAME" \
                --yes \
                --output none
        fi

        # ----------------------------------------------------
        # Attempt VM creation
        # ----------------------------------------------------

        if az vm create \
            --resource-group "$RG" \
            --name "$VM_NAME" \
            --location "$LOCATION" \
            --zone "$ZONE" \
            --size "$VM_SIZE" \
            --image "$IMAGE" \
            --nics "$NIC_NAME" \
            --os-disk-name "$OS_DISK_NAME" \
            --admin-username "$ADMIN_USERNAME" \
            --admin-password "$ADMIN_PASSWORD" \
            --authentication-type password \
            --security-type TrustedLaunch \
            --enable-secure-boot true \
            --enable-vtpm true \
            --os-disk-delete-option Delete \
            --output none
        then

            echo ""
            echo "============================================================"
            echo "SUCCESS"
            echo "VM: $VM_NAME"
            echo "Availability Zone: $ZONE"
            echo "============================================================"

            return 0

        else

            echo ""
            echo "FAILED: $VM_NAME in Availability Zone $ZONE"
            echo "Retrying next Availability Zone..."

            # ------------------------------------------------
            # Cleanup OS disk created by failed VM attempt
            # ------------------------------------------------

            if az disk show \
                --resource-group "$RG" \
                --name "$OS_DISK_NAME" \
                --output none 2>/dev/null
            then

                az disk delete \
                    --resource-group "$RG" \
                    --name "$OS_DISK_NAME" \
                    --yes \
                    --output none
            fi

        fi

    done

    echo ""
    echo "============================================================"
    echo "FAILURE"
    echo "VM: $VM_NAME"
    echo "Zones attempted: 1, 2, 3"
    echo "============================================================"

    return 1
}

# ============================================================
# 15. Create WEB VM
# ============================================================

if ! create_vm_with_retry \
    "$WEB_VM" \
    "$WEB_NIC" \
    "$WEB_OS_DISK"
then

    echo "ERROR: WEB-VM-IT-PROD could not be created in Zones 1, 2, or 3."
    exit 1
fi

# ============================================================
# 16. Verify WEB VM Exists
# ============================================================

echo "Verifying WEB VM..."

az vm show \
    --resource-group "$RG" \
    --name "$WEB_VM" \
    --output none

# ============================================================
# 17. Create DB VM
# ============================================================

if ! create_vm_with_retry \
    "$DB_VM" \
    "$DB_NIC" \
    "$DB_OS_DISK"
then

    echo "ERROR: DB-VM-IT-PROD could not be created in Zones 1, 2, or 3."
    exit 1
fi

# ============================================================
# 18. Verify DB VM Exists
# ============================================================

echo "Verifying DB VM..."

az vm show \
    --resource-group "$RG" \
    --name "$DB_VM" \
    --output none

# ============================================================
# 19. Get Resource IDs
# ============================================================

WEB_NIC_ID=$(az network nic show \
    --resource-group "$RG" \
    --name "$WEB_NIC" \
    --query "id" \
    --output tsv)

DB_NIC_ID=$(az network nic show \
    --resource-group "$RG" \
    --name "$DB_NIC" \
    --query "id" \
    --output tsv)

WEB_PIP_ID=$(az network public-ip show \
    --resource-group "$RG" \
    --name "$WEB_PIP" \
    --query "id" \
    --output tsv)

DB_PIP_ID=$(az network public-ip show \
    --resource-group "$RG" \
    --name "$DB_PIP" \
    --query "id" \
    --output tsv)

WEB_NSG_ID=$(az network nsg show \
    --resource-group "$RG" \
    --name "$WEB_NSG" \
    --query "id" \
    --output tsv)

DB_NSG_ID=$(az network nsg show \
    --resource-group "$RG" \
    --name "$DB_NSG" \
    --query "id" \
    --output tsv)

# ============================================================
# 20. Verify WEB NIC attached to WEB VM
# ============================================================

echo "Verifying WEB NIC attachment..."

if ! az vm show \
    --resource-group "$RG" \
    --name "$WEB_VM" \
    --query "networkProfile.networkInterfaces[].id" \
    --output tsv | grep -Fxq "$WEB_NIC_ID"
then
    echo "ERROR: $WEB_NIC is not attached to $WEB_VM."
    exit 1
fi

# ============================================================
# 21. Verify DB NIC attached to DB VM
# ============================================================

echo "Verifying DB NIC attachment..."

if ! az vm show \
    --resource-group "$RG" \
    --name "$DB_VM" \
    --query "networkProfile.networkInterfaces[].id" \
    --output tsv | grep -Fxq "$DB_NIC_ID"
then
    echo "ERROR: $DB_NIC is not attached to $DB_VM."
    exit 1
fi

# ============================================================
# 22. Verify WEB OS Disk
# ============================================================

echo "Verifying WEB OS Disk..."

WEB_ATTACHED_DISK=$(az vm show \
    --resource-group "$RG" \
    --name "$WEB_VM" \
    --query "storageProfile.osDisk.name" \
    --output tsv)

if [ "$WEB_ATTACHED_DISK" != "$WEB_OS_DISK" ]; then
    echo "ERROR: WEB OS Disk verification failed."
    exit 1
fi

# ============================================================
# 23. Verify DB OS Disk
# ============================================================

echo "Verifying DB OS Disk..."

DB_ATTACHED_DISK=$(az vm show \
    --resource-group "$RG" \
    --name "$DB_VM" \
    --query "storageProfile.osDisk.name" \
    --output tsv)

if [ "$DB_ATTACHED_DISK" != "$DB_OS_DISK" ]; then
    echo "ERROR: DB OS Disk verification failed."
    exit 1
fi

# ============================================================
# 24. Verify WEB Public IP -> WEB NIC
# ============================================================

echo "Verifying WEB Public IP association..."

WEB_ATTACHED_PIP_ID=$(az network nic show \
    --resource-group "$RG" \
    --name "$WEB_NIC" \
    --query "ipConfigurations[0].publicIPAddress.id" \
    --output tsv)

if [ "$WEB_ATTACHED_PIP_ID" != "$WEB_PIP_ID" ]; then
    echo "ERROR: WEB Public IP is not associated with WEB NIC."
    exit 1
fi

# ============================================================
# 25. Verify DB Public IP -> DB NIC
# ============================================================

echo "Verifying DB Public IP association..."

DB_ATTACHED_PIP_ID=$(az network nic show \
    --resource-group "$RG" \
    --name "$DB_NIC" \
    --query "ipConfigurations[0].publicIPAddress.id" \
    --output tsv)

if [ "$DB_ATTACHED_PIP_ID" != "$DB_PIP_ID" ]; then
    echo "ERROR: DB Public IP is not associated with DB NIC."
    exit 1
fi

# ============================================================
# 26. Verify WEB NSG -> WEB Subnet
# ============================================================

echo "Verifying WEB NSG subnet association..."

WEB_SUBNET_NSG_ID=$(az network vnet subnet show \
    --resource-group "$RG" \
    --vnet-name "$VNET" \
    --name "$WEB_SUBNET" \
    --query "networkSecurityGroup.id" \
    --output tsv)

if [ "$WEB_SUBNET_NSG_ID" != "$WEB_NSG_ID" ]; then
    echo "ERROR: WEB NSG is not correctly associated with WEB subnet."
    exit 1
fi

# ============================================================
# 27. Verify DB NSG -> DB Subnet
# ============================================================

echo "Verifying DB NSG subnet association..."

DB_SUBNET_NSG_ID=$(az network vnet subnet show \
    --resource-group "$RG" \
    --vnet-name "$VNET" \
    --name "$DB_SUBNET" \
    --query "networkSecurityGroup.id" \
    --output tsv)

if [ "$DB_SUBNET_NSG_ID" != "$DB_NSG_ID" ]; then
    echo "ERROR: DB NSG is not correctly associated with DB subnet."
    exit 1
fi

# ============================================================
# 28. Verify WEB NIC has NO Direct NSG
# ============================================================

echo "Verifying WEB NIC has no direct NSG..."

WEB_NIC_NSG=$(az network nic show \
    --resource-group "$RG" \
    --name "$WEB_NIC" \
    --query "networkSecurityGroup.id" \
    --output tsv)

if [ -n "$WEB_NIC_NSG" ]; then
    echo "ERROR: WEB NIC has a direct NSG association."
    exit 1

# ============================================================
# 29. Verify DB NIC has NO Direct NSG
# ============================================================

echo "Verifying DB NIC has no direct NSG..."

DB_NIC_NSG=$(az network nic show \
    --resource-group "$RG" \
    --name "$DB_NIC" \
    --query "networkSecurityGroup.id" \
    --output tsv)

if [ -n "$DB_NIC_NSG" ]; then

# ============================================================

# ============================================================

    --resource-group "$RG" \
    --name "$WEB_VM" \
    --query "zones[0]" \
    --output tsv)

if [[ "$WEB_ZONE" != "1" && "$WEB_ZONE" != "2" && "$WEB_ZONE" != "3" ]]; then
    echo "ERROR: Invalid WEB VM Availability Zone."
    exit 1
fi

# ============================================================
# 31. Verify DB VM Availability Zone
# ============================================================

DB_ZONE=$(az vm show \
    --resource-group "$RG" \
    --name "$DB_VM" \
    --query "zones[0]" \
    --output tsv)

if [[ "$DB_ZONE" != "1" && "$DB_ZONE" != "2" && "$DB_ZONE" != "3" ]]; then

    --output tsv)

    --resource-group "$RG" \

    --output tsv)

echo ""
