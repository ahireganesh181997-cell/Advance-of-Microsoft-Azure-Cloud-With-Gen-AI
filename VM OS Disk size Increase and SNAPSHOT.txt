 #!/bin/bash
set -euo pipefail

# Variables
RG="RG-IT-PROD"
VM="VM-IT-PROD"
TARGET_SIZE=512

# Retrieve the existing OS disk name
OS_DISK=$(az vm show \
  --resource-group "$RG" \
  --name "$VM" \
  --query "storageProfile.osDisk.name" \
  --output tsv)

if [[ -z "$OS_DISK" || "$OS_DISK" == "None" ]]; then
  echo "ERROR: Unable to retrieve the OS disk for VM $VM." >&2
  exit 1
fi

# Retrieve the existing OS disk size
CURRENT_SIZE=$(az disk show \
  --resource-group "$RG" \
  --name "$OS_DISK" \
  --query "diskSizeGb" \
  --output tsv)

echo "OS Disk: $OS_DISK"
echo "Current OS disk size: ${CURRENT_SIZE} GiB"

# Resize the existing OS disk only if needed
if [[ "$CURRENT_SIZE" -ne "$TARGET_SIZE" ]]; then
  if [[ "$CURRENT_SIZE" -gt "$TARGET_SIZE" ]]; then
    echo "ERROR: OS disk size cannot be reduced from ${CURRENT_SIZE} GiB to ${TARGET_SIZE} GiB." >&2
    exit 1
  fi

  az disk update \
    --resource-group "$RG" \
    --name "$OS_DISK" \
    --size-gb "$TARGET_SIZE" \
    --output none
fi

# Verify the OS disk size
VERIFIED_SIZE=$(az disk show \
  --resource-group "$RG" \
  --name "$OS_DISK" \
  --query "diskSizeGb" \
  --output tsv)

if [[ "$VERIFIED_SIZE" -eq "$TARGET_SIZE" ]]; then
  echo "SUCCESS: OS disk '$OS_DISK' resized and verified at ${TARGET_SIZE} GiB."
else
  echo "ERROR: OS disk resize verification failed. Actual size: ${VERIFIED_SIZE} GiB." >&2
  exit 1
fi

#!/bin/bash
set -euo pipefail

# Variables
RG="RG-IT-PROD"
VM="VM-IT-PROD"
SNAPSHOT="VM-IT-PROD-SNAPSHOT"

# Retrieve the OS disk ID associated with the VM
OS_DISK_ID=$(az vm show \
  --resource-group "$RG" \
  --name "$VM" \
  --query "storageProfile.osDisk.managedDisk.id" \
  --output tsv)

if [[ -z "$OS_DISK_ID" || "$OS_DISK_ID" == "None" ]]; then
  echo "ERROR: Unable to retrieve the OS disk ID for VM '$VM'." >&2
  exit 1
fi

# Create an incremental snapshot of the existing OS disk
az snapshot create \
  --resource-group "$RG" \
  --name "$SNAPSHOT" \
  --source "$OS_DISK_ID" \
  --incremental true \
  --output none

# Verify that the snapshot exists and is incremental
SNAPSHOT_DETAILS=$(az snapshot show \
  --resource-group "$RG" \
  --name "$SNAPSHOT" \
  --query "{name:name, provisioningState:provisioningState, incremental:incremental, source:creationData.sourceResourceId}" \
  --output json)

PROVISIONING_STATE=$(az snapshot show \
  --resource-group "$RG" \
  --name "$SNAPSHOT" \
  --query "provisioningState" \
  --output tsv)

INCREMENTAL=$(az snapshot show \
  --resource-group "$RG" \
  --name "$SNAPSHOT" \
  --query "incremental" \
  --output tsv)

if [[ "$PROVISIONING_STATE" == "Succeeded" && "$INCREMENTAL" == "true" ]]; then
  echo "SUCCESS: Incremental snapshot '$SNAPSHOT' created and verified."
  echo "$SNAPSHOT_DETAILS"
else
  echo "ERROR: Snapshot verification failed." >&2
  echo "$SNAPSHOT_DETAILS" >&2
  exit 1
fi
